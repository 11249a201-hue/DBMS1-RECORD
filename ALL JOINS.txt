CREATE TABLE STUDENT
(
 STUDENT_ID NUMBER(5) PRIMARY KEY,
 STUDENT_NAME VARCHAR2(30),
 MARKS NUMBER(3),
 DEPT_ID NUMBER(3),
 CONSTRAINT FK_DEPT FOREIGN KEY (DEPT_ID)
 REFERENCES DEPARTMENT(DEPT_ID)
);

SUBQUERY – Students Scoring Above Average Marks
SELECT STUDENT_NAME, MARKS
FROM STUDENT
WHERE MARKS >
(
 SELECT AVG(MARKS)
 FROM STUDENT
);

SUBQUERY WITH IN – Students in the CSE Department
SELECT STUDENT_NAME
FROM STUDENT
WHERE DEPT_ID IN
(
 SELECT DEPT_ID
 FROM DEPARTMENT
 WHERE DEPT_NAME = 'CSE'
);

CORRELATED SUBQUERY – Students with Maximum Marks in Their Department
SELECT STUDENT_NAME, MARKS, DEPT_ID
FROM STUDENT S1
WHERE MARKS =
(
 SELECT MAX(MARKS)
 FROM STUDENT S2
 WHERE S1.DEPT_ID = S2.DEPT_ID
);
